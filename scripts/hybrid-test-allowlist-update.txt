-- hybrid-test-allowlist-update.txt
-- Populates the hybrid_negotiation test allowlist while preserving existing settings.
-- Keeps the global hybrid flag disabled.

DO $$
DECLARE
  allowlist_jsonb JSONB := to_jsonb(ARRAY[
    'e0d535e5-e90f-4a4c-b3f2-6937634517c7',
    'e0727e87-2fd6-40ab-bdb5-eba509145901',
    'd0e8e8e8-e8e8-e8e8-e8e8-d0e8e8e8e8e8'
  ]::text[]);
  services_jsonb JSONB := '["shop", "errand"]'::jsonb;
BEGIN
  -- Update existing row if present, preserving all other JSON fields
  UPDATE public.marketplace_settings
  SET value = jsonb_set(
                jsonb_set(
                  jsonb_set(
                    jsonb_set(
                      COALESCE(value, '{}'::jsonb),
                      '{allowlist}', allowlist_jsonb, true
                    ),
                    '{enabled}', 'false'::jsonb, true
                  ),
                  '{enabledServices}', services_jsonb, true
                ),
                '{eligibleServices}', services_jsonb, true
              ),
      updated_at = now()
  WHERE key = 'hybrid_negotiation'
    AND tenant_id IS NULL;

  -- Insert safe default if row does not exist
  IF NOT FOUND THEN
    INSERT INTO public.marketplace_settings (tenant_id, key, value, updated_at)
    VALUES (
      NULL,
      'hybrid_negotiation',
      jsonb_build_object(
        'enabled', false,
        'maxRounds', 3,
        'timeoutSeconds', 120,
        'maxDriverAttempts', 5,
        'claimTimeoutSeconds', 60,
        'enabledServices', services_jsonb,
        'eligibleServices', services_jsonb,
        'rideMinimumDistanceKm', 30,
        'makeOfferEnabled', true,
        'acceptFareEnabled', true,
        'allowlist', allowlist_jsonb
      ),
      now()
    );
  END IF;

  -- Update system_configs only if a hybrid allowlist key is already present
  UPDATE public.system_configs
  SET value = allowlist_jsonb::text,
      updated_at = now()
  WHERE key = 'hybrid_allowlist';
END $$;
