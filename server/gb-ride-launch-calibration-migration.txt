-- Make shadow mode explicit before the strategy is created or updated. This
-- does not enable market pricing and prevents live application if it is enabled elsewhere.
INSERT INTO public.marketplace_settings (tenant_id, key, value)
SELECT NULL, 'market_pricing_shadow_mode', 'true'::jsonb
WHERE NOT EXISTS (
    SELECT 1 FROM public.marketplace_settings
    WHERE tenant_id IS NULL AND key = 'market_pricing_shadow_mode'
);

UPDATE public.marketplace_settings
SET value = 'true'::jsonb,
    updated_at = now()
WHERE key = 'market_pricing_shadow_mode';

INSERT INTO public.marketplace_settings (tenant_id, key, value)

WHERE NOT EXISTS (
    SELECT 1 FROM public.marketplace_settings
    WHERE tenant_id IS NULL AND key = 'market_pricing_enabled'
);

UPDATE public.marketplace_settings
SET value = 'false'::jsonb,
    updated_at = now()
WHERE key = 'market_pricing_enabled';

       'beat_market', 8.00, c.launch_target,

WITH services(service_type) AS (
    VALUES ('ride'), ('errand'), ('delivery'), ('van-moving')
)
INSERT INTO public.pricing_config (
    service_type, country_code, market_city, zone_id,
    base_fare, per_km, per_min, service_fee, minimum_fare,
    currency_code, is_active
)
SELECT s.service_type, 'GB', NULL, NULL, 0, 0, 0, 0, 0, 'GBP', TRUE
FROM services s
WHERE NOT EXISTS (
    SELECT 1 FROM public.pricing_config p
    WHERE p.service_type = s.service_type
      AND COALESCE(p.country_code, 'GB') = 'GB'
      AND NULLIF(BTRIM(COALESCE(p.market_city, '')), '') IS NULL
      AND NULLIF(BTRIM(COALESCE(p.zone_id, '')), '') IS NULL
);
